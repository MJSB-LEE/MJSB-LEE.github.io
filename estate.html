/** * Site Visit (Lawatan Ladang) — sync to Google Drive * Paste this code at https://script.google.com (New project), then Deploy as a Web App. */ // 1) CHANGE this secret word. Type the same word in the app: Settings → Google Drive. const SECRET = 'CHANGE_THIS_SECRET'; // 2) Name of the main folder in Google Drive const ROOT_NAME = 'Site Visit'; function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (d.key !== SECRET) return out({ ok: false, error: 'Wrong secret word' }); if (d.action === 'ping') return out({ ok: true }); if (d.action === 'trash') { // photo folder attached to a visit or deleted in the app const props = PropertiesService.getScriptProperties(), fid = props.getProperty('v_' + d.visitId); if (fid) { try { DriveApp.getFolderById(fid).setTrashed(true); } catch (e) {} props.deleteProperty('v_' + d.visitId); } return out({ ok: true }); } const folder = visitFolder(d.visitId, d.folderName); if (d.action === 'file') { const old = folder.getFilesByName(d.name); while (old.hasNext()) old.next().setTrashed(true); folder.createFile(Utilities.newBlob(Utilities.base64Decode(d.data), d.mime, d.name)); } else if (d.action === 'cleanup') { // remove photo files that were deleted in the app const keep = {}; d.keep.forEach(n => keep[n] = true); const it = folder.getFiles(); while (it.hasNext()) { const f = it.next(); if (!keep[f.getName()]) f.setTrashed(true); } } return out({ ok: true, folderUrl: folder.getUrl() }); } catch (err) { return out({ ok: false, error: String(err) }); } } function doGet() { return out({ ok: true, info: 'Site Visit sync is running' }); } function visitFolder(id, name) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const props = PropertiesService.getScriptProperties(); const fid = props.getProperty('v_' + id); if (fid) { try { const f = DriveApp.getFolderById(fid); if (!f.isTrashed()) { if (f.getName() !== name) f.setName(name); return f; } } catch (e) { /* folder was deleted — make a new one */ } } const f = rootFolder().createFolder(name); props.setProperty('v_' + id, f.getId()); return f; } finally { lock.releaseLock(); } } function rootFolder() { const it = DriveApp.getFoldersByName(ROOT_NAME); return it.hasNext() ? it.next() : DriveApp.createFolder(ROOT_NAME); } function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
